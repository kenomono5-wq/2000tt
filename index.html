    // ---------- IndexedDB cache ----------

    let idb;

    function openIDB(){

      return new Promise((resolve,reject)=>{

        const req = indexedDB.open(DB_NAME, 1);

        req.onupgradeneeded = e => {

          const d = e.target.result;

          if(!d.objectStoreNames.contains(STORE_A)) d.createObjectStore(STORE_A, {keyPath:"id"});

          if(!d.objectStoreNames.contains(STORE_R)) d.createObjectStore(STORE_R, {keyPath:"id"});

        };

        req.onsuccess = ()=>resolve(req.result);

        req.onerror = ()=>reject(req.error);

      });

    }

    function idbGetAll(store){

      return new Promise((resolve,reject)=>{

        const tx=idb.transaction(store,"readonly"), req=tx.objectStore(store).getAll();

        req.onsuccess=()=>resolve(req.result||[]); req.onerror=()=>reject(req.error);

      });

    }

    function idbPut(store,val){

      return new Promise((resolve,reject)=>{

        const tx=idb.transaction(store,"readwrite");

        tx.objectStore(store).put(val);

        tx.oncomplete=()=>resolve();

        tx.onerror=()=>reject(tx.error);

      });

    }

    function idbDelete(store,key){

      return new Promise((resolve,reject)=>{

        const tx=idb.transaction(store,"readwrite");

        tx.objectStore(store).delete(key);

        tx.oncomplete=()=>resolve();

        tx.onerror=()=>reject(tx.error);

      });

    }

    async function initCloud(){

      try{

        app = initializeApp(firebaseConfig);

        cloud = getFirestore(app);

        await seedIfEmptyCloud();


        unsubAthletes = onSnapshot(collection(cloud,"athletes"), async snap => {

          athletes = snap.docs.map(d=>d.data());

          for(const a of athletes) await idbPut(STORE_A, a);

          refreshAll();

        });


        unsubRecords = onSnapshot(collection(cloud,"records"), async snap => {

          records = snap.docs.map(d=>d.data());

          for(const r of records) await idbPut(STORE_R, r);

          refreshAll();

        });


        setCloudBadge("クラウド接続中（リアルタイム同期）", true);

      } catch(e){

        console.error(e);

        setCloudBadge("クラウド未接続（キャッシュ表示）", false);

      }

    }

    // ---------- Derived ----------

    function enriched(){

      const amap = new Map(athletes.map(a=>[a.id,a]));

      return records.map(r=>{

        const a = amap.get(r.athleteId);

        if(!a) return null;

        const yi = yearInfo(a.entranceYear);

        return {

          ...r,

          athleteName:a.name,

          gender:a.gender,

          side:a.side,

          status:yi.status,

          yearLabel:yi.yearLabel,

          split500:Number(r.timeInSeconds)/4,

          idt:calcIDT(a.gender, Number(r.weight), Number(r.timeInSeconds))

        };

      }).filter(Boolean).sort((x,y)=>new Date(x.date)-new Date(y.date));

    }

    function renderRanking(){

      const gender=$("fGender").value, status=$("fStatus").value, side=$("fSide").value, sort=$("fSort").value, pb=$("fPB").checked;

      let arr = enriched().filter(r=>{

        if(gender!=="ALL" && r.gender!==gender) return false;

        if(status==="ACTIVE" && r.status!=="現役") return false;

        if(!sideMatch(r.side, side)) return false;

        return true;

      });

      if(pb){

        const m=new Map();

        for(const r of arr){

          const p=m.get(r.athleteId);

          if(!p) m.set(r.athleteId,r);

          else{

            const better = sort==="TIME" ? r.timeInSeconds<p.timeInSeconds : r.idt>p.idt;

            if(better) m.set(r.athleteId,r);

          }

        }

        arr=[...m.values()];

      }

      arr.sort((a,b)=>sort==="TIME" ? a.timeInSeconds-b.timeInSeconds : b.idt-a.idt);


      $("rankBody").innerHTML = arr.length ? arr.map((r,i)=>`

        <tr class="border-b border-slate-800/70">

          <td class="p-2 font-bold ${i<3?'text-amber-300':''}">${i+1}</td><td class="p-2 font-semibold">${r.athleteName}</td><td class="p-2">${r.gender}</td>

          <td class="p-2">${r.status}${r.status==="現役"?`(${r.yearLabel})`:''}</td><td class="p-2">${r.side}</td><td class="p-2 text-cyan-300 font-bold">${fmtTime(r.timeInSeconds)}</td>

          <td class="p-2">${fmtTime(r.split500)}</td><td class="p-2">${r.weight.toFixed(1)}</td><td class="p-2 text-amber-300 font-bold">${r.idt.toFixed(2)}</td>

          <td class="p-2">${fmtDate(r.date)}</td><td class="p-2">${r.event}</td>

        </tr>

      `).join("") : `<tr><td colspan="11" class="p-5 text-center text-slate-400">データなし</td></tr>`;

    }


    function renderChronology(){

      const arr = enriched();

      const best = new Map(), breaker = new Map();

      arr.forEach(r=>{

        const p=best.get(r.athleteId);

        if(p===undefined){ best.set(r.athleteId,r.timeInSeconds); breaker.set(r.id,false); }

        else if(r.timeInSeconds<p){ best.set(r.athleteId,r.timeInSeconds); breaker.set(r.id,true); }

        else breaker.set(r.id,false);

      });


      const g = new Map();

      arr.forEach(r=>{ if(!g.has(r.event)) g.set(r.event,[]); g.get(r.event).push(r); });


      const cards = [...g.entries()].map(([event,list])=>{

        list.sort((a,b)=>new Date(a.date)-new Date(b.date));

        return {

          event, ds:list[0].date, de:list[list.length-1].date,

          avg:list.reduce((s,x)=>s+x.timeInSeconds,0)/list.length,

          maxIdt:Math.max(...list.map(x=>x.idt)),

          count:new Set(list.map(x=>x.athleteId)).size,

          breakers:[...new Set(list.filter(x=>breaker.get(x.id)).map(x=>x.athleteName))]

        };

      }).sort((a,b)=>new Date(b.de)-new Date(a.de));


      $("timeline").innerHTML = `<div class="timeline-line"></div>` + (cards.length ? cards.map(c=>`

        <article class="relative">

          <div class="timeline-dot"></div>

          <div class="glass rounded-xl p-4">

            <div class="flex flex-col md:flex-row md:justify-between gap-2">

              <h4 class="font-bold text-cyan-300">${c.event}</h4>

              <p class="text-xs text-slate-400">${fmtDate(c.ds)}${c.ds===c.de?'':' 〜 '+fmtDate(c.de)}</p>

            </div>

            <div class="grid grid-cols-2 lg:grid-cols-4 gap-2 mt-3">

              <div class="glass rounded-lg p-3"><p class="text-xs text-slate-400">平均タイム</p><p class="font-bold">${fmtTime(c.avg)}</p></div>

              <div class="glass rounded-lg p-3"><p class="text-xs text-slate-400">最高IDT%</p><p class="font-bold text-amber-300">${c.maxIdt.toFixed(2)}%</p></div>

              <div class="glass rounded-lg p-3"><p class="text-xs text-slate-400">参加人数</p><p class="font-bold">${c.count}名</p></div>

              <div class="glass rounded-lg p-3"><p class="text-xs text-slate-400">新記録達成者</p><p class="font-semibold ${c.breakers.length?'text-emerald-300':'text-slate-400'}">${c.breakers.length?c.breakers.join("、"):"なし"}</p></div>

            </div>

          </div>

        </article>

      `).join("") : `<div class="glass rounded-xl p-4">データなし</div>`);

    }

        e.preventDefault();

        const fd = new FormData(e.target);

          name:String(fd.get("name")).trim(),

          gender:fd.get("gender"),

          side:fd.get("side"),

        if(!athlete.name){ alert("氏名を入力してください"); return; }

        }catch(err){ alert("保存失敗: "+err.message); }

          athleteId:fd.get("athleteId"),

          date:fd.get("date"),

          event:String(fd.get("event")).trim(),

          timeInSeconds:Number(sec),

          memo:String(fd.get("memo")||"").trim()

        };

        try{

          await setDoc(doc(cloud,"records",rec.id), rec);

          await idbPut(STORE_R, rec);

          e.target.reset(); closeModal("recordModal");

        }catch(err){ alert("保存失敗: "+err.message); }

      });

    }
